// https://github.com/KudryavtsevDmitry/ht2-rev/blob/master/src/sound/directsound.cpp
// Snd_CreateBuffer
//
int __fastcall Snd_CreateBuffer(PlayObject *self)

{
  BOOL createResult;
  HRESULT lockResult;
  uint copyCount;
  DWORD *pSrc;
  DWORD *pDst1;
  DWORD *pDst2;
  int exceptionCode1;
  int exceptionCode2;
  int successFlag;
  LPVOID lockPtr1;
  LPVOID lockPtr2;
  DWORD lockBytes2;
  DWORD lockBytes1;
  BYTE *savedEsp;
  void *prevExceptionList;
  BYTE *sehHandler;
  int sehState;
  IDirectSoundBuffer *l_pDsBuffer;
  IDirectSoundBuffer **ppDsBuffer;

  sehHandler = &SehHandler_Snd_CreateBuffer;
  prevExceptionList = ExceptionList;
  savedEsp = &stack0xffffffc0;
  successFlag = 1;
  sehState = 0;
  ExceptionList = &prevExceptionList;
  createResult = Mixer_CreateBuffer(g_pDirectSound,g_soundMixerTable[self->soundIndex].format,
                                    (IDirectSoundBuffer *)&self->pDsBuffer,self->dataSize);
  if (createResult == 0) {
    exceptionCode2 = 0;
                    /* WARNING: Subroutine does not return */
    __CxxThrowException@8(&exceptionCode2,&DAT_00668478);
  }
  l_pDsBuffer = self->pDsBuffer;
  lockResult = (*l_pDsBuffer->lpVtbl->Lock)
                         (l_pDsBuffer,0,self->dataSize,&lockPtr1,&lockBytes1,&lockPtr2,&lockBytes2,0
                         );
  if (lockResult == 0) {
    pSrc = self->data;
    pDst1 = lockPtr1;
    for (copyCount = lockBytes1 >> 2; copyCount != 0; copyCount = copyCount - 1) {
      *pDst1 = *pSrc;
      pSrc = pSrc + 1;
      pDst1 = pDst1 + 1;
    }
    for (copyCount = lockBytes1 & 3; copyCount != 0; copyCount = copyCount - 1) {
      *(char *)pDst1 = (char)*pSrc;
      pSrc = (DWORD *)((int)pSrc + 1);
      pDst1 = (DWORD *)((int)pDst1 + 1);
    }
    if (lockBytes2 != 0) {
      pSrc = (DWORD *)((int)self->data + lockBytes1);
      pDst2 = lockPtr2;
      for (copyCount = lockBytes2 >> 2; copyCount != 0; copyCount = copyCount - 1) {
        *pDst2 = *pSrc;
        pSrc = pSrc + 1;
        pDst2 = pDst2 + 1;
      }
      for (copyCount = lockBytes2 & 3; copyCount != 0; copyCount = copyCount - 1) {
        *(char *)pDst2 = (char)*pSrc;
        pSrc = (DWORD *)((int)pSrc + 1);
        pDst2 = (DWORD *)((int)pDst2 + 1);
      }
    }
    (*self->pDsBuffer->lpVtbl->Unlock)(self->pDsBuffer,lockPtr1,lockBytes1,lockPtr2,lockBytes2);
    (*self->pDsBuffer->lpVtbl->SetPan)(self->pDsBuffer,self->pan);
    (*self->pDsBuffer->lpVtbl->SetVolume)(self->pDsBuffer,self->volume);
    (*self->pDsBuffer->lpVtbl->Play)(self->pDsBuffer,0,0,self->looping != 0);
    ExceptionList = prevExceptionList;
    return successFlag;
  }
  exceptionCode1 = 0;
                    /* WARNING: Subroutine does not return */
  __CxxThrowException@8(&exceptionCode1,&DAT_00668478);
}

// https://github.com/KudryavtsevDmitry/ht2-rev/blob/master/src/sound/directsound.cpp
// Mixer_CreateBuffer
//
BOOL __cdecl
Mixer_CreateBuffer(IDirectSound *pDirectSound,tWAVEFORMATEX *pFormat,IDirectSoundBuffer *pDsBuffer,
                  DWORD dataSize)

{
  HRESULT bufferCreateResult;
  DWORD i;
  DWORD *pInitializerIter;
  DSBUFFERDESC dsBufferDesc;

  i = 9;
  pInitializerIter = &dsBufferDesc.dwSize;
  for (; i != 0; i = i - 1) {
    *pInitializerIter = 0;
    pInitializerIter = pInitializerIter + 1;
  }
  dsBufferDesc.dwBufferBytes = dataSize;
  dsBufferDesc.lpwfxFormat = pFormat;
                    /* sizeof(DSBUFFERDESC) */
  dsBufferDesc.dwSize = 0x24;
                    /* DSBCAPS_CTRLVOLUME | DSBCAPS_CTRLPAN | DSBCAPS_CTRLFREQUENCY */
  dsBufferDesc.dwFlags = 0xe0;
  bufferCreateResult =
       (*pDirectSound->lpVtbl->CreateSoundBuffer)(pDirectSound,&dsBufferDesc,pDsBuffer,0);
  if (bufferCreateResult == 0) {
    return TRUE;
  }
  pDsBuffer->lpVtbl = (IDirectSoundBufferVtbl *)NULL;
  return FALSE;
}
